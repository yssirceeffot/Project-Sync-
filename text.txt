        // WEB AUDIO API SOUND SYNTHESIZER
        let audioCtx = null;

        function getAudioContext() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            return audioCtx;
        }

        function playSound(type) {
            try {
                const ctx = getAudioContext();
                const now = ctx.currentTime;

                if (type === 'status_note') {
                    // Soft bubble pop
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.exponentialRampToValueAtTime(880, now + 0.12);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(now);
                    osc.stop(now + 0.12);

                } else if (type === 'task_add') {
                    // Metallic click chime
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(587.33, now);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(now);
                    osc.stop(now + 0.15);

                } else if (type === 'file_add') {
                    // Upward dual-tone sweep
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.exponentialRampToValueAtTime(659.25, now + 0.18);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.18);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(now);
                    osc.stop(now + 0.18);

                } else if (type === 'task_check') {
                    // Mid completion chime
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(659.25, now);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(now);
                    osc.stop(now + 0.2);

                } else if (type === 'milestone_complete') {
                    // Grand C-Major Victory Fanfare
                    const notes = [523.25, 659.25, 783.99, 1046.50];
                    notes.forEach((freq, idx) => {
                        const osc = ctx.createOscillator();
                        const gain = ctx.createGain();
                        const noteTime = now + (idx * 0.09);
                        osc.type = 'triangle';
                        osc.frequency.setValueAtTime(freq, noteTime);
                        gain.gain.setValueAtTime(0.25, noteTime);
                        gain.gain.exponentialRampToValueAtTime(0.01, noteTime + 0.35);
                        osc.connect(gain);
                        gain.connect(ctx.destination);
                        osc.start(noteTime);
                        osc.stop(noteTime + 0.35);
                    });

                } else if (type === 'chat_msg') {
                    // Crisp double message pop
                    [0, 0.08].forEach((delay, idx) => {
                        const osc = ctx.createOscillator();
                        const gain = ctx.createGain();
                        const msgTime = now + delay;
                        osc.type = 'sine';
                        osc.frequency.setValueAtTime(800 + (idx * 200), msgTime);
                        osc.frequency.exponentialRampToValueAtTime(400, msgTime + 0.06);
                        gain.gain.setValueAtTime(0.18, msgTime);
                        gain.gain.exponentialRampToValueAtTime(0.01, msgTime + 0.06);
                        osc.connect(gain);
                        gain.connect(ctx.destination);
                        osc.start(msgTime);
                        osc.stop(msgTime + 0.06);
                    });
                }
            } catch (e) {
                console.warn('Audio playback error', e);
            }
        }

                    <span style="color:#64748b;">Created by ${serverData.admin}</span>
                `;
            }

            setInterval(checkExpiredNotes, 10000);
        });

        function initWorkspace(mode) {
            getAudioContext();

            if (mode === 'create') {
                const adminName = document.getElementById('adminNameInput').value.trim();
                const serverName = document.getElementById('serverNameInput').value.trim();
                const serverBio = document.getElementById('serverBioInput').value.trim();

                if (!adminName || !serverName) {
                    alert('Please enter your name and space name!');
                    return;
                }

                serverData = { name: serverName, bio: serverBio || 'Team Project Workspace', admin: adminName };
                currentUser = { name: adminName, role: 'Space Admin', isAdmin: true };
            } else {
                const joinerName = document.getElementById('joinerNameInput').value.trim();
                const joinerRole = document.getElementById('joinerRoleInput').value.trim();

                if (!joinerName) {
                    alert('Please enter your name to join!');
                    return;
                }

                currentUser = { name: joinerName, role: joinerRole || 'Member', isAdmin: false };
                
                if (serverData.admin) {
                    addMemberToList(serverData.admin, 'Space Admin', true);
                }
            }

            document.getElementById('onboardingScreen').style.display = 'none';

            document.getElementById('displayServerTitle').innerText = serverData.name;
            document.getElementById('displayServerBio').innerText = serverData.bio;

            addMemberToList(currentUser.name, currentUser.role, currentUser.isAdmin);

            window.location.hash = `server=${encodeURIComponent(serverData.name)}&admin=${encodeURIComponent(serverData.admin)}`;
            playSound('task_add');
        }

        function addMemberToList(name, role, isAdmin) {
            const initials = name.substring(0,2).toUpperCase();
            const id = 'mem-' + Date.now() + Math.random().toString(36).substr(2,4);
            
            members.push({ name, role, isAdmin });

            const card = document.createElement('div');
            card.className = 'member-card';
            card.id = id;
            card.innerHTML = `
                <div class="member-left">
                    <div class="story-ring"><div class="avatar">${initials}</div></div>
                    <div>
                        <div style="font-size: 0.85rem; font-weight:700;">${name}</div>
                        <span class="custom-role-badge ${isAdmin ? 'admin-badge' : ''}" onclick="editRole(this)">${role}</span>
                    </div>
                </div>
                ${!isAdmin ? `<button class="remove-btn" onclick="removeEl('${id}')"><i class="fa-solid fa-user-minus"></i></button>` : ''}
            `;
            document.getElementById('memberList').appendChild(card);
            renderDMUserList();
        }

        function copySpaceLink() {
            const shareUrl = `${window.location.origin}${window.location.pathname}#server=${encodeURIComponent(serverData.name)}&admin=${encodeURIComponent(serverData.admin)}`;
            navigator.clipboard.writeText(shareUrl);
            
            const alertBox = document.getElementById('copyAlert');
            alertBox.style.display = 'block';
            setTimeout(() => alertBox.style.display = 'none', 3000);
            playSound('task_add');
        }

        function editRole(badgeEl) {
            const newRole = prompt("Enter custom role tag for this member:");
            if (newRole && newRole.trim() !== "") badgeEl.innerText = newRole.trim();
        }

        function removeEl(id) {
            const el = document.getElementById(id);
            if (el) el.remove();
        }

        function toggleForm(id) {
            const el = document.getElementById(id);
            el.style.display = el.style.display === 'flex' ? 'none' : 'flex';
        }

        function addFileLink() {
            const nameInp = document.getElementById('fileNameInp');
            const urlInp = document.getElementById('fileUrlInp');
            if (nameInp.value.trim() !== "") {
                const emptyMsg = document.getElementById('emptyFileMsg');
                if (emptyMsg) emptyMsg.remove();

                const list = document.getElementById('fileHubList');
                const card = document.createElement('div');
                card.className = 'file-item-card';
                card.innerHTML = `
                    <div>
                        <i class="fa-solid fa-link" style="color:#2563eb; margin-right:6px;"></i>
                        <strong>${nameInp.value.trim()}</strong>
                    </div>
                    <a href="${urlInp.value.trim() || '#'}" target="_blank" style="color:var(--primary-blue); font-size:0.7rem; font-weight:bold; text-decoration:none;">Open</a>
                `;
                list.appendChild(card);
                nameInp.value = '';
                urlInp.value = '';
                toggleForm('fileForm');
                playSound('file_add');
            }
        }

        function addTask() {
            const inp = document.getElementById('taskInp');
            if (inp.value.trim() !== '') {
                totalTasks++;
                const div = document.createElement('div');
                div.className = 'task-row';
                div.innerHTML = `<input type="checkbox" onchange="updateProgress(this)"> <span>${inp.value}</span>`;
                document.getElementById('taskList').appendChild(div);
                inp.value = '';
                calculateProgress();
                playSound('task_add');
            }
        }

        function updateProgress(cb) {
            if (cb.checked) {
                completedTasks++;
                playSound('task_check');
            } else {
                completedTasks--;
            }
            calculateProgress();
        }

        function calculateProgress() {
            if (totalTasks === 0) return;
            const pct = Math.round((completedTasks / totalTasks) * 100);
            document.getElementById('progressPercent').innerText = `${pct}%`;
            document.getElementById('progressBarFill').style.width = `${pct}%`;

            if (pct === 100 && completedTasks > 0) {
                playSound('milestone_complete');
            }
        }

        function addNote() {
            const inp = document.getElementById('noteInp');
            if (inp.value.trim() !== '') {
                const initials = currentUser.name ? currentUser.name.substring(0,2).toUpperCase() : 'ME';
                const noteId = 'note-' + Date.now();
                const createdAt = Date.now();

                const item = document.createElement('div');
                item.className = 'note-item';
                item.id = noteId;
                item.setAttribute('data-timestamp', createdAt);
                item.innerHTML = `
                    <div class="note-bubble">
                        ${inp.value}
                        <button class="delete-note-btn" onclick="deleteNote('${noteId}')" title="Delete note">&times;</button>
                    </div>
                    <div class="story-ring"><div class="avatar">${initials}</div></div>
                `;
                
                document.getElementById('notesBar').insertBefore(item, document.getElementById('notesBar').firstChild);
                statusNotes.push({ id: noteId, timestamp: createdAt });
                inp.value = '';
                playSound('status_note');
            }
        }

        function deleteNote(id) {
            const noteEl = document.getElementById(id);
            if (noteEl) {
                noteEl.remove();
                statusNotes = statusNotes.filter(n => n.id !== id);
            }
        }

        function checkExpiredNotes() {
            const now = Date.now();
            const twentyFourHoursMs = 24 * 60 * 60 * 1000;

            statusNotes.forEach(note => {
                if (now - note.timestamp >= twentyFourHoursMs) {
                    deleteNote(note.id);
                }
            });
        }

        let activeTab = 'group';
        let currentDMUser = null;

        function toggleChat() {
            const win = document.getElementById('chatWin');
            win.style.display = win.style.display === 'flex' ? 'none' : 'flex';
        }

        function switchTab(t) {
            activeTab = t;
            document.getElementById('gTab').classList.toggle('active', t === 'group');
            document.getElementById('pTab').classList.toggle('active', t === 'private');
            
            if (t === 'group') {
                document.getElementById('gMsg').style.display = 'flex';
                document.getElementById('pMsg').style.display = 'none';
                document.getElementById('activeDMMsg').style.display = 'none';
                currentDMUser = null;
            } else {
                document.getElementById('gMsg').style.display = 'none';
                document.getElementById('pMsg').style.display = 'flex';
                document.getElementById('activeDMMsg').style.display = 'none';
            }
        }

        function renderDMUserList() {
            const box = document.getElementById('dmUserList');
            box.innerHTML = '';
            members.forEach(m => {
                if (m.name !== currentUser.name) {
                    const initials = m.name.substring(0,2).toUpperCase();
                    const item = document.createElement('div');
                    item.className = 'dm-user-item';
                    item.onclick = () => openDM(m.name);
                    item.innerHTML = `
                        <div class="story-ring"><div class="avatar">${initials}</div></div>
                        <div><div style="font-weight:bold; font-size:0.8rem;">${m.name}</div><div style="font-size:0.68rem; color:var(--duo-green);">${m.role}</div></div>
                    `;
                    box.appendChild(item);
                }
            });
        }

        function openDM(userName) {
            currentDMUser = userName;
            document.getElementById('pMsg').style.display = 'none';
            document.getElementById('activeDMMsg').style.display = 'flex';
            document.getElementById('dmTargetName').innerText = 'DM with ' + userName;
        }

        function backToDMList() {
            currentDMUser = null;
            document.getElementById('activeDMMsg').style.display = 'none';
            document.getElementById('pMsg').style.display = 'flex';
        }

        function sendMsg() {
            const inp = document.getElementById('chatInp');
            if (inp.value.trim() !== '') {
                const msg = document.createElement('div');
                msg.className = 'chat-msg mine';
                msg.innerHTML = `<strong>You:</strong> ${inp.value}`;
                let targetBox = (activeTab === 'group') ? document.getElementById('gMsg') : (currentDMUser ? document.getElementById('activeDMMsg') : document.getElementById('pMsg'));
                targetBox.appendChild(msg);
                targetBox.scrollTop = targetBox.scrollHeight;
                inp.value = '';
                playSound('chat_msg');
            }
        }

        let t = 0;
        setInterval(() => {
            t++;
            let h = String(Math.floor(t / 3600)).padStart(2, '0');
            let m = String(Math.floor((t % 3600) / 60)).padStart(2, '0');
            let s = String(t % 60).padStart(2, '0');
            document.getElementById('timer').innerText = `${h}:${m}:${s}`;
        }, 1000);
